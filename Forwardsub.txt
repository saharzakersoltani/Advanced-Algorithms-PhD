بخش اول: تعریف تابع

function x = forwardsub(L,b)

[m,n] = size(L);

if m ~= n
    error('Matrix L is not square!')
end

x = zeros(n,1);

x(1) = b(1)/L(1,1);

for i = 2:n

    s = 0;

    for j = 1:i-1
        s = s + L(i,j)*x(j);
    end

    if L(i,i) == 0
        error('Matrix L is singular')
    end

    x(i) = (b(i)-s)/L(i,i);

end

end
