بخش اول: تعریف تابع

function x = gauss(A,b)

n = length(b);

Ab = [A b];

for k = 1:n-1

    for i = k+1:n

        m = Ab(i,k)/Ab(k,k);

        Ab(i,:) = Ab(i,:) - m*Ab(k,:);

    end

end

U = Ab(:,1:n);

c = Ab(:,n+1);

x = backsub(U,c);

end
