تعریف تابع:بخش اول

function y = backsub(T,b)

[m,n] = size(T);

if m ~= n
    error('Matrix T is not square!')
end

y = zeros(n,1);

for i = n:-1:1

    s = 0;

    if i ~= n
        s = T(i,i+1:n)*y(i+1:n);
    end

    if T(i,i) == 0
        error('Matrix T is singular')
    end

    y(i) = (b(i)-s)/T(i,i);

end

end
