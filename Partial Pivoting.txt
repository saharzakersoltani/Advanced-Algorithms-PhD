بخش اول: تعریف تابع

function x = gausspivot(A,b)

n = length(b);

Ab = [A b];

for k = 1:n-1

    [~,p] = max(abs(Ab(k:n,k)));

    p = p + k - 1;

    if Ab(p,k) == 0
        error('Matrix is singular')
    end

    if p ~= k
        temp = Ab(k,:);
        Ab(k,:) = Ab(p,:);
        Ab(p,:) = temp;
    end

    for i = k+1:n

        m = Ab(i,k)/Ab(k,k);

        Ab(i,:) = Ab(i,:) - m*Ab(k,:);

    end

end

U = Ab(:,1:n);

c = Ab(:,n+1);

x = backsub(U,c);

end
