function x = gausscomplete(A,b)

[n,m] = size(A);

if n ~= m
    error('A must be a square matrix');
end

if length(b) ~= n
    error('Dimensions of A and b do not match');
end

b = b(:);

Ab = [A b];

col_order = 1:n;

for k = 1:n

    S = abs(Ab(k:n,k:n));

    [~,idx] = max(S(:));

    [r,c] = ind2sub(size(S),idx);

    r = r + k - 1;
    c = c + k - 1;

    if r ~= k
        temp = Ab(k,:);
        Ab(k,:) = Ab(r,:);
        Ab(r,:) = temp;
    end

    if c ~= k
        temp = Ab(:,k);
        Ab(:,k) = Ab(:,c);
        Ab(:,c) = temp;

        temp = col_order(k);
        col_order(k) = col_order(c);
        col_order(c) = temp;
    end

    if abs(Ab(k,k)) < 1e-12
        error('A is singular or nearly singular');
    end

    Ab(k,:) = Ab(k,:) / Ab(k,k);

    for i = k+1:n

        factor = Ab(i,k);

        Ab(i,:) = Ab(i,:) - factor*Ab(k,:);

    end

end

x_temp = zeros(n,1);

for i = n:-1:1

    x_temp(i) = Ab(i,n+1) ...
        - Ab(i,i+1:n)*x_temp(i+1:n);

end

x = zeros(n,1);

x(col_order) = x_temp;

end
